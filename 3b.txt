    <p id="add"></p>

    <button onclick="ClickFunction()">
        Click here!
    </button>
    <script>
        function ClickFunction() {
            let result = confirm("Press OK to close this option");
            if (result === true) {

                document.getElementById("add").textContent =
                    "User clicked OK";
                console.log("User clicked OK");
            } else {
                document.getElementById("add").textContent =
                    "User clicked Cancel";
                console.log("User clicked Cancel");
            }
        }
    </script>
